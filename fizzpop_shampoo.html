Step 1 : Database Connection

เชื่อมต่อ Google Sheets (Web App URL)

วางลิงก์ Web App URL จาก Google Apps Script เพื่อบันทึกคำสั่งซื้อและอัปเดตสถานะการแพ็คลง Google Sheets อัตโนมัติ

* หากยังไม่ได้กรอก ระบบจะทำงานในโหมดทดสอบ (บันทึกลงเบราว์เซอร์ LocalStorage) ได้ทันที

Step 2 : Backend Script

โค้ด Google Apps Script (คัดลอกไปวางใน Google Sheet)

สร้างแท็บชื่อ Orders และตั้งหัวคอลัมน์ A1:I1 ตามลำดับ แล้ว Deploy เป็น Web App (สิทธิ์: Anyone)

const SHEET_NAME = 'Orders';

// รับคำสั่งซื้อจากหน้าร้าน (POST)
function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  const data = JSON.parse(e.postData.contents);
  
  const orderId = data.orderId || ('ORD-' + Math.floor(100000 + Math.random() * 900000));
  const timestamp = new Date().toLocaleString('th-TH', { timeZone: 'Asia/Bangkok' });
  
  sheet.appendRow([
    orderId,
    timestamp,
    data.name,
    "'" + data.phone,
    data.address,
    data.itemsText,
    data.total,
    data.payment,
    'รอแพ็คสินค้า'
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({ status: 'success', orderId: orderId }))
    .setMimeType(ContentService.MimeType.JSON);
}

// ดึงรายการออเดอร์ & อัปเดตสถานะแพ็คสินค้า (GET)
function doGet(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  const action = e.parameter.action;
  
  if (action === 'updateStatus') {
    const orderId = e.parameter.orderId;
    const rows = sheet.getDataRange().getValues();
    for (let i = 1; i < rows.length; i++) {
      if (rows[i][0] === orderId) {
        sheet.getRange(i + 1, 9).setValue('แพ็คแล้ว / พร้อมส่ง');
        break;
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: 'updated' }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  const rows = sheet.getDataRange().getValues();
  const headers = rows[0];
  const orders = rows.slice(1).map(row => {
    let obj = {};
    headers.forEach((h, i) => obj[h] = row[i]);
    return obj;
  });
  
  return ContentService.createTextOutput(JSON.stringify(orders.reverse()))
    .setMimeType(ContentService.MimeType.JSON);
}
